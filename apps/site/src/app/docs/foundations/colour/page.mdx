import {
  LEVELS, MINIMUM_RATIO, PALETTE_HUES, apcaContrast, auditContrast, contrastRatio,
} from '@rtkelly13/design-system';
import { Post, postMetadata } from '@/components/docs/Post';

export const metadata = postMetadata('/docs/foundations/colour');

export const checks = auditContrast(LEVELS);
export const failing = checks.filter((c) => !c.passes).length;

{/* Data helpers: they compute each figure's data from package exports. Later posts keep such helpers at the top in the same way. */}

export const hueHalves = [PALETTE_HUES.slice(0, 5), PALETTE_HUES.slice(5)];
export const hueColumns = (hues) => hues.map((name) => ({ name }));
export const hueRows = (level, hues) => [
  { label: 'Base', values: hues.map((h) => LEVELS[level].palette[h]) },
  { label: 'Bright', values: hues.map((h) => LEVELS[level].paletteBright[h]) },
];

export const roles = [
  ['text.primary', (d) => d.text.primary],
  ['text.secondary', (d) => d.text.secondary],
  ['text.muted', (d) => d.text.muted],
  ['intent.info', (d) => d.intent.info],
  ['intent.success', (d) => d.intent.success],
  ['intent.warning', (d) => d.intent.warning],
  ['intent.danger', (d) => d.intent.danger],
];
export const roleRows = (level) => {
  const def = LEVELS[level];
  return roles.map(([token, pick]) => ({
    token,
    after: pick(def),
    values: [
      `${contrastRatio(pick(def), def.surface.sunken).toFixed(2)}:1`,
      `${Math.round(Math.abs(apcaContrast(pick(def), def.surface.sunken)))}`,
    ],
  }));
};

## Ten hues, drawn twice

Each Level authors its own ten hues. They share hue angles, so the two read as one brand, and they share no values. A base hue clears {MINIMUM_RATIO.palette}:1 against the tightest ground of its Level. A bright hue carries emphasis, so it clears {MINIMUM_RATIO.paletteBright}:1.

## Measured in CI

The contrast gate audits every pair in both Levels. One failing pair fails the check.
